// Nécessite le function isAdmin() déjà défini dans les règles TACLAR.
// IMPORTANT : ne pas publier avant mise à jour du parcours de compte
// et test avec un compte de démonstration distinct.

match /registrationRequests/{requestId} {
  function isVerifiedOwner() {
    return request.auth != null
      && request.auth.token.email_verified == true
      && request.auth.uid == requestId;
  }

  // Un dossier maximum par compte Firebase Auth (UID document).
  // Une nouvelle tentative ne crée donc pas de doublon du même compte.
  allow create: if isVerifiedOwner()
    && request.resource.data.keys().hasOnly([
      "ownerUid", "parentName", "email", "phone",
      "childName", "age", "schoolId", "classId",
      "otherClass", "transportRequested", "pickupArea",
      "status", "createdAt"
    ])
    && request.resource.data.ownerUid == request.auth.uid
    && request.resource.data.email == request.auth.token.email
    && request.resource.data.status == "pending"
    && request.resource.data.createdAt == request.time
    && request.resource.data.parentName is string
    && request.resource.data.parentName.size() > 1
    && request.resource.data.parentName.size() <= 100
    && request.resource.data.childName is string
    && request.resource.data.childName.size() > 1
    && request.resource.data.childName.size() <= 100
    && request.resource.data.phone is string
    && request.resource.data.phone.size() >= 8
    && request.resource.data.phone.size() <= 30
    && request.resource.data.age is string
    && request.resource.data.age.size() <= 3
    && request.resource.data.schoolId in [
      "elite-international",
      "college-capitaine-ntchorere"
    ]
    && request.resource.data.classId is string
    && request.resource.data.classId.size() <= 80
    && request.resource.data.otherClass is string
    && request.resource.data.otherClass.size() <= 60
    && request.resource.data.transportRequested is bool
    && request.resource.data.pickupArea is string
    && request.resource.data.pickupArea.size() <= 130;

  // Parent : voir uniquement SON statut. Admin : consulter tout.
  allow get: if isAdmin() || (
    request.auth != null && request.auth.uid == requestId
  );
  allow list: if isAdmin();

  // Admin peut modifier le statut uniquement, sans altérer les
  // champs d'identité. La création d'accès scolaires reste distincte.
  allow update: if isAdmin()
    && request.resource.data.diff(resource.data).affectedKeys()
      .hasOnly(["status", "reviewedAt"])
    && request.resource.data.status in [
      "pending", "correction_requested", "approved",
      "rejected", "archived"
    ]
    && request.resource.data.reviewedAt == request.time;

  // Suppression désactivée à ce stade : un processus maîtrisé de
  // clôture/effacement sera ajouté avec contrôle de rattachement.
  allow delete: if false;
}
